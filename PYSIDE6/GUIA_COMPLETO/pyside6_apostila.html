Introdução e Instalação
O que é PySide6, por que usar e como configurar o ambiente

PySide6 é o binding oficial do Qt para Python, mantido pela The Qt Company. Permite criar aplicações desktop multiplataforma — Windows, Linux e macOS — usando Python puro. É gratuito sob licença LGPL para uso comercial e educacional.

🖥️
Multiplataforma
Um único código roda no Windows, Linux e macOS sem alterações.
🐍
100% Python
Todo o código fica em Python. Qt Designer gera arquivos que você importa.
🎨
Qt Designer
Editor visual drag-and-drop para montar interfaces sem escrever código.
⚡
Signals & Slots
Sistema de eventos reativo e desacoplado, o coração do Qt.

Instalação

  1. Certifique-se de ter o Python 3.10+ instalado (python --version)
  2. Instale o PySide6 com pip
  3. Verifique a instalação com um import simples
  4. Abra o Qt Designer (já vem embutido!)
bash terminal 
# Instalar PySide6 (inclui Qt Designer e pyside6-uic)
pip install pyside6

# Verificar versão instalada
python -c "import PySide6; print(PySide6.__version__)"

# Abrir o Qt Designer
pyside6-designer

# Converter arquivo .ui para .py (ver seção 9)
pyside6-uic minha_janela.ui -o minha_janela_ui.py
ℹ️
PyQt6 vs PySide6: As duas bibliotecas são quase idênticas na API — ambas fazem binding para Qt6. A diferença principal é a licença: PySide6 é LGPL (mais liberal), PyQt6 é GPL/Comercial. Para fins educacionais, ambas funcionam igual.

Primeira Janela
QApplication, QMainWindow e QWidget — a estrutura mínima

Toda aplicação PySide6 precisa de três coisas: uma QApplication (única, gerencia o loop de eventos), uma janela principal (geralmente QMainWindow ou QWidget), e o app.exec() que mantém a aplicação rodando.

Exemplo 1 — Janela mais simples possível

python 01_janela_minima.py 
import sys
from PySide6.QtWidgets import QApplication, QMainWindow

# 1. Criar a aplicação (recebe sys.argv para argumentos de linha de comando)
app = QApplication(sys.argv)

# 2. Criar a janela principal
janela = QMainWindow()
janela.setWindowTitle("Minha Primeira Janela")
janela.resize(600, 400)   # largura x altura em pixels
janela.show()             # exibir a janela

# 3. Iniciar o loop de eventos (a aplicação fica viva aqui)
sys.exit(app.exec())

Exemplo 2 — Janela com classe (forma correta para projetos reais)

Usar herança de QMainWindow é a forma padrão de organizar o código. Tudo que compõe a janela fica no __init__.

python 02_janela_classe.py 
import sys
from PySide6.QtWidgets import QApplication, QMainWindow, QLabel
from PySide6.QtCore import Qt

class MinhaJanela(QMainWindow):
    """Janela principal da aplicação."""

    def __init__(self):
        # Sempre chame o construtor da classe-pai primeiro!
        super().__init__()

        # Configurações da janela
        self.setWindowTitle("PySide6 — Aula 02")
        self.setMinimumSize(400, 300)   # tamanho mínimo
        self.resize(640, 480)           # tamanho inicial

        # Widget central — QMainWindow EXIGE um widget central
        label = QLabel("Olá, PySide6! 🎉")
        label.setAlignment(Qt.AlignCenter)  # centralizar texto

        # Definir o widget central da janela
        self.setCentralWidget(label)


# Ponto de entrada padrão Python
if __name__ == "__main__":
    app = QApplication(sys.argv)
    janela = MinhaJanela()
    janela.show()
    sys.exit(app.exec())
💡
QMainWindow vs QWidget: Use QMainWindow quando precisar de menu, barra de ferramentas ou barra de status. Use QWidget para janelas simples ou sub-janelas. QMainWindow sempre precisa de um setCentralWidget().

Exemplo 3 — Propriedades da janela

python 03_propriedades_janela.py 
import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from PySide6.QtGui import QIcon
from PySide6.QtCore import Qt

class Janela(QMainWindow):
    def __init__(self):
        super().__init__()

        self.setWindowTitle("Propriedades da Janela")

        # Tamanho fixo (impede redimensionamento)
        # self.setFixedSize(640, 480)

        # Tamanho com mínimo e máximo
        self.setMinimumSize(400, 300)
        self.setMaximumSize(1200, 800)
        self.resize(640, 480)

        # Centralizar a janela na tela
        self.centralizar()

        # Ícone (coloque um arquivo .ico ou .png na mesma pasta)
        # self.setWindowIcon(QIcon("icone.png"))

        # Remover botões de maximizar/minimizar (opcional)
        # self.setWindowFlags(Qt.Window | Qt.WindowCloseButtonHint)

    def centralizar(self):
        """Centraliza a janela na tela do monitor."""
        tela = self.screen().availableGeometry()
        x = (tela.width()  - self.width())  // 2
        y = (tela.height() - self.height()) // 2
        self.move(x, y)

if __name__ == "__main__":
    app = QApplication(sys.argv)
    janela = Janela()
    janela.show()
    sys.exit(app.exec())

Widgets Essenciais
QLabel, QPushButton, QLineEdit, QTextEdit e mais

Widgets são os blocos de construção da interface gráfica. Cada elemento visual (botão, caixa de texto, rótulo) é um widget. Abaixo os mais usados:

WidgetUsoMétodo principal

Exemplo — Vários widgets numa janela

python 04_widgets_basicos.py 
import sys
from PySide6.QtWidgets import (
    QApplication, QWidget, QVBoxLayout,
    QLabel, QPushButton, QLineEdit,
    QTextEdit, QCheckBox
)
from PySide6.QtCore import Qt

class WidgetsDemo(QWidget):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("Widgets Básicos")
        self.resize(420, 500)

        # Layout vertical para empilhar os widgets
        layout = QVBoxLayout()
        self.setLayout(layout)

        # ── QLabel ──────────────────────────────────────────
        titulo = QLabel("Cadastro de Aluno")
        titulo.setAlignment(Qt.AlignCenter)
        # Estilo inline (subset de CSS)
        titulo.setStyleSheet("font-size: 18px; font-weight: bold; color: #00b4d8; padding: 8px;")
        layout.addWidget(titulo)

        # ── QLineEdit ───────────────────────────────────────
        self.campo_nome = QLineEdit()
        self.campo_nome.setPlaceholderText("Digite seu nome completo...")
        layout.addWidget(QLabel("Nome:"))
        layout.addWidget(self.campo_nome)

        # QLineEdit com modo senha (esconde os caracteres)
        self.campo_senha = QLineEdit()
        self.campo_senha.setPlaceholderText("Senha...")
        self.campo_senha.setEchoMode(QLineEdit.EchoMode.Password)
        layout.addWidget(QLabel("Senha:"))
        layout.addWidget(self.campo_senha)

        # ── QTextEdit ───────────────────────────────────────
        layout.addWidget(QLabel("Observações:"))
        self.area_texto = QTextEdit()
        self.area_texto.setPlaceholderText("Digite observações aqui...")
        self.area_texto.setMaximumHeight(80)
        layout.addWidget(self.area_texto)

        # ── QCheckBox ───────────────────────────────────────
        self.check_ativo = QCheckBox("Aluno ativo")
        self.check_ativo.setChecked(True)  # marcado por padrão
        layout.addWidget(self.check_ativo)

        # ── QPushButton ─────────────────────────────────────
        btn_salvar = QPushButton("💾 Salvar")
        btn_salvar.setStyleSheet("""
            QPushButton {
                background-color: #0077b6;
                color: white;
                padding: 8px 20px;
                border-radius: 6px;
                font-size: 14px;
            }
            QPushButton:hover { background-color: #00b4d8; }
            QPushButton:pressed { background-color: #005f8d; }
        """)
        btn_salvar.clicked.connect(self.salvar)
        layout.addWidget(btn_salvar)

        # Label de resultado
        self.label_resultado = QLabel("")
        self.label_resultado.setAlignment(Qt.AlignCenter)
        layout.addWidget(self.label_resultado)

    def salvar(self):
        """Lê os valores dos widgets e exibe no label."""
        nome   = self.campo_nome.text()
        obs    = self.area_texto.toPlainText()
        ativo  = "Sim" if self.check_ativo.isChecked() else "Não"

        if not nome:
            self.label_resultado.setStyleSheet("color: #f85149;")
            self.label_resultado.setText("⚠️ Preencha o nome!")
            return

        self.label_resultado.setStyleSheet("color: #3fb950;")
        self.label_resultado.setText(f"✅ Salvo: {nome} | Ativo: {ativo}")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    w = WidgetsDemo()
    w.show()
    sys.exit(app.exec())
💡
StyleSheet: PySide6 aceita um subconjunto de CSS para estilizar widgets. Use widget.setStyleSheet("...") ou defina estilos globais na aplicação com app.setStyleSheet(...). Suporta seletores como QPushButton:hover.

Layouts
Organizar widgets com QVBoxLayout, QHBoxLayout e QGridLayout

Layouts são gerenciadores de posicionamento. Nunca use coordenadas fixas (x,y) — os layouts se adaptam ao redimensionamento da janela automaticamente.

⬇️
QVBoxLayout
Empilha widgets verticalmente, um abaixo do outro.
➡️
QHBoxLayout
Posiciona widgets horizontalmente, lado a lado.
⊞
QGridLayout
Grade com linhas e colunas — pode ocupar múltiplas células.
📋
QFormLayout
Pares label + campo, ideal para formulários.

Exemplo — QVBoxLayout e QHBoxLayout

python 05_layouts_vbox_hbox.py 
import sys
from PySide6.QtWidgets import (
    QApplication, QWidget, QVBoxLayout, QHBoxLayout,
    QLabel, QPushButton, QLineEdit
)

class LayoutDemo(QWidget):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("Layouts — VBox + HBox")
        self.resize(480, 300)

        # Layout principal: VERTICAL
        layout_principal = QVBoxLayout()
        self.setLayout(layout_principal)

        # Título no topo
        layout_principal.addWidget(QLabel("📦 Controle de Estoque"))

        # Campo de nome + botão numa linha horizontal
        layout_nome = QHBoxLayout()
        layout_nome.addWidget(QLabel("Produto:"))
        self.campo_produto = QLineEdit()
        layout_nome.addWidget(self.campo_produto)

        # Adiciona o layout horizontal DENTRO do vertical
        layout_principal.addLayout(layout_nome)

        # Linha de quantidade
        layout_qtd = QHBoxLayout()
        layout_qtd.addWidget(QLabel("Qtd:"))
        self.campo_qtd = QLineEdit()
        self.campo_qtd.setMaximumWidth(80)
        layout_qtd.addWidget(self.campo_qtd)
        layout_qtd.addStretch()  # empurra tudo para a esquerda
        layout_principal.addLayout(layout_qtd)

        # Espaçador para empurrar botões para baixo
        layout_principal.addStretch()

        # Linha de botões no rodapé
        layout_botoes = QHBoxLayout()
        layout_botoes.addStretch()  # empurra botões para a direita
        btn_cancelar = QPushButton("Cancelar")
        btn_adicionar = QPushButton("✚ Adicionar")
        btn_adicionar.clicked.connect(self.adicionar)
        layout_botoes.addWidget(btn_cancelar)
        layout_botoes.addWidget(btn_adicionar)
        layout_principal.addLayout(layout_botoes)

        self.label_status = QLabel("")
        layout_principal.addWidget(self.label_status)

    def adicionar(self):
        prod = self.campo_produto.text()
        qtd  = self.campo_qtd.text()
        self.label_status.setText(f"Adicionado: {prod} x {qtd}")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    w = LayoutDemo()
    w.show()
    sys.exit(app.exec())

Exemplo — QGridLayout (calculadora básica)

python 06_grid_layout.py 
import sys
from PySide6.QtWidgets import (
    QApplication, QWidget, QGridLayout,
    QPushButton, QLineEdit
)

class Calculadora(QWidget):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("Calculadora")
        self.setFixedSize(260, 320)

        layout = QGridLayout()
        self.setLayout(layout)

        # Display no topo — ocupa 4 colunas (col 0 até 3)
        self.display = QLineEdit("0")
        self.display.setReadOnly(True)
        self.display.setStyleSheet("font-size: 22px; padding: 6px; text-align: right;")
        layout.addWidget(self.display, 0, 0, 1, 4)  # row, col, rowspan, colspan

        # Mapa de botões: (texto, linha, coluna)
        botoes = [
            ("7", 1, 0), ("8", 1, 1), ("9", 1, 2), ("÷", 1, 3),
            ("4", 2, 0), ("5", 2, 1), ("6", 2, 2), ("×", 2, 3),
            ("1", 3, 0), ("2", 3, 1), ("3", 3, 2), ("−", 3, 3),
            ("0", 4, 0), (".",  4, 1), ("=", 4, 2), ("+", 4, 3),
        ]

        for texto, linha, coluna in botoes:
            btn = QPushButton(texto)
            btn.setMinimumHeight(50)
            btn.clicked.connect(lambda checked, t=texto: self.clicar(t))
            layout.addWidget(btn, linha, coluna)

    def clicar(self, texto):
        # Lógica simplificada para demonstrar o grid
        atual = self.display.text()
        if texto == "=":
            try:
                # Substituir símbolos pelos operadores Python
                expr = atual.replace("÷", "/").replace("×", "*").replace("−", "-")
                resultado = str(eval(expr))
                self.display.setText(resultado)
            except:
                self.display.setText("Erro")
        elif atual == "0" and texto not in ".+-×÷−":
            self.display.setText(texto)
        else:
            self.display.setText(atual + texto)

if __name__ == "__main__":
    app = QApplication(sys.argv)
    w = Calculadora()
    w.show()
    sys.exit(app.exec())

Signals e Slots
O coração do Qt — comunicação desacoplada entre objetos

O sistema de Signals & Slots é o mecanismo de eventos do Qt. Um signal é emitido quando algo acontece (botão clicado, texto mudou, valor alterado). Um slot é uma função que responde a esse signal. A conexão é feita com .connect().

Widget
signal
 .connect() →
Função
slot
  emite → executa

Exemplo — Signals básicos

python 08_signals_slots.py 
import sys
from PySide6.QtWidgets import (
    QApplication, QWidget, QVBoxLayout, QHBoxLayout,
    QLabel, QPushButton, QLineEdit, QSlider, QSpinBox
)
from PySide6.QtCore import Qt, Signal

class SignalsDemo(QWidget):
    # ── Signal customizado (opcional, avançado) ──────────
    # Pode definir seus próprios signals com a classe Signal
    nome_alterado = Signal(str)  # emite uma string

    def __init__(self):
        super().__init__()
        self.setWindowTitle("Signals & Slots")
        self.resize(460, 400)
        layout = QVBoxLayout(self)

        # ── Exemplo 1: clicked (sem argumento) ───────────
        layout.addWidget(QLabel("── Botão ──────────────"))
        self.contador = 0
        self.label_contador = QLabel("Cliques: 0")
        btn = QPushButton("Clique em mim!")

        # .clicked é um Signal; passamos uma função como slot
        btn.clicked.connect(self.incrementar)
        layout.addWidget(btn)
        layout.addWidget(self.label_contador)

        # ── Exemplo 2: textChanged (emite a string atual) ─
        layout.addWidget(QLabel("── Campo de texto ──────"))
        campo = QLineEdit()
        self.label_eco = QLabel("Você digitou: ")

        # textChanged emite o texto atual a cada tecla
        campo.textChanged.connect(self.mostrar_texto)
        layout.addWidget(campo)
        layout.addWidget(self.label_eco)

        # ── Exemplo 3: valueChanged (emite um int) ────────
        layout.addWidget(QLabel("── Slider ──────────────"))
        h = QHBoxLayout()
        self.slider = QSlider(Qt.Horizontal)
        self.slider.setRange(0, 100)
        self.label_slider = QLabel("50")
        self.slider.setValue(50)

        # valueChanged emite o valor atual como inteiro
        self.slider.valueChanged.connect(self.atualizar_slider)
        h.addWidget(self.slider)
        h.addWidget(self.label_slider)
        layout.addLayout(h)

        # ── Exemplo 4: Signal customizado ────────────────
        layout.addWidget(QLabel("── Signal customizado ──"))
        campo2 = QLineEdit()
        self.label_custom = QLabel("Aguardando...")

        # Conectar nosso próprio signal a um slot
        self.nome_alterado.connect(self.receber_nome)
        # Emitir o signal customizado quando o texto mudar
        campo2.textChanged.connect(self.nome_alterado)
        layout.addWidget(campo2)
        layout.addWidget(self.label_custom)

    # ── SLOTS (funções que respondem aos signals) ─────────

    def incrementar(self):
        """Slot do botão — chamado sem argumentos."""
        self.contador += 1
        self.label_contador.setText(f"Cliques: {self.contador}")

    def mostrar_texto(self, texto: str):
        """Slot do textChanged — recebe o texto atual."""
        self.label_eco.setText(f"Você digitou: {texto}")

    def atualizar_slider(self, valor: int):
        """Slot do valueChanged — recebe o novo valor."""
        self.label_slider.setText(str(valor))

    def receber_nome(self, nome: str):
        """Slot do nosso signal customizado."""
        self.label_custom.setText(f"Signal recebido: '{nome}'")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    w = SignalsDemo()
    w.show()
    sys.exit(app.exec())
ℹ️
Tabela de signals comuns:
QPushButton.clicked → sem argumento
QLineEdit.textChanged → str
QLineEdit.returnPressed → sem argumento (Enter)
QCheckBox.stateChanged → int (0=desmarcado, 2=marcado)
QComboBox.currentIndexChanged → int
QSlider.valueChanged → int
QSpinBox.valueChanged → int

Exemplo — Desconectar e múltiplas conexões

python 09_multi_conexoes.py 
import sys
from PySide6.QtWidgets import QApplication, QWidget, QVBoxLayout, QPushButton, QLabel

class MultiConexao(QWidget):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("Múltiplas conexões")
        layout = QVBoxLayout(self)

        btn = QPushButton("Clique")
        self.label1 = QLabel("Log 1: —")
        self.label2 = QLabel("Log 2: —")

        # Um signal pode ter MÚLTIPLOS slots conectados
        btn.clicked.connect(self.slot_a)
        btn.clicked.connect(self.slot_b)

        layout.addWidget(btn)
        layout.addWidget(self.label1)
        layout.addWidget(self.label2)

        btn_off = QPushButton("Desconectar slot_a")
        btn_off.clicked.connect(lambda: btn.clicked.disconnect(self.slot_a))
        layout.addWidget(btn_off)

    def slot_a(self):
        self.label1.setText("Log 1: slot_a executou!")

    def slot_b(self):
        self.label2.setText("Log 2: slot_b executou!")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    w = MultiConexao()
    w.show()
    sys.exit(app.exec())

import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from PySide6.QtCore import QFile
from PySide6.uic import loadUi   # ← importação do carregador

class MinhaJanela(QMainWindow):
    def __init__(self):
        super().__init__()

        # Carregar o arquivo .ui e aplicar NESTA instância (self)
        # O arquivo deve estar na mesma pasta ou use o caminho completo
        loadUi("janela_principal.ui", self)

        # A partir daqui, todos os objectNames do .ui são
        # atributos de self — acesse-os diretamente!

        # self.label_titulo   → o QLabel que chamamos "label_titulo" no Designer
        # self.input_nome     → o QLineEdit "input_nome"
        # self.btn_salvar     → o QPushButton "btn_salvar"

        # Conectar signal do botão ao nosso slot
        self.btn_salvar.clicked.connect(self.salvar)

    def salvar(self):
        nome = self.input_nome.text()
        if nome:
            self.label_titulo.setText(f"Olá, {nome}! 👋")
        else:
            self.label_titulo.setText("⚠️ Digite um nome!")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    janela = MinhaJanela()
    janela.show()

Objetos Nomeados no Backend
Como o objectName do Qt Designer vira atributo Python

Esta é a seção mais importante para quem vai usar o Qt Designer. O objectName que você define no Designer (no painel de propriedades) se torna automaticamente um atributo de self no backend, após o loadUi() ou setupUi().

⚠️
Regra de ouro: O nome do atributo Python é exatamente o objectName definido no Qt Designer. Se no Designer o campo se chama lineEdit_nome, no Python você acessa com self.lineEdit_nome. Erros de digitação causam AttributeError.

import sys
from PySide6.QtWidgets import QApplication, QMainWindow, QTableWidgetItem
from PySide6.uic import loadUi

class CadastroAluno(QMainWindow):
    """
    Backend da tela de cadastro de alunos.
    Os objectNames do .ui são acessados como self.NOME.
    """

    def __init__(self):
        super().__init__()
        loadUi("cadastro_aluno.ui", self)

        # ─────────────────────────────────────────────────
        # Após loadUi(), TODOS os widgets nomeados no Designer
        # se tornam atributos de self automaticamente.
        # ─────────────────────────────────────────────────

        # Configurações iniciais dos widgets
        self.comboBox_curso.addItems(["Python", "MySQL", "Redes", "Web Dev"])
        self.spinBox_idade.setRange(14, 80)
        self.tableWidget_alunos.setColumnCount(4)
        self.tableWidget_alunos.setHorizontalHeaderLabels(
            ["Nome", "Email", "Idade", "Curso"]
        )
        self.tableWidget_alunos.horizontalHeader().setStretchLastSection(True)

        # Conectar signals usando os objectNames como atributos
        self.pushButton_salvar.clicked.connect(self.salvar_aluno)
        self.pushButton_limpar.clicked.connect(self.limpar_campos)
        self.lineEdit_nome.textChanged.connect(self.validar_formulario)
        self.lineEdit_email.textChanged.connect(self.validar_formulario)

        # Estado inicial
        self.validar_formulario()

    # ── Slots ─────────────────────────────────────────────

    def salvar_aluno(self):
        """Lê os widgets nomeados e insere na tabela."""
        # Acessar valor de cada widget pelo objectName
        nome   = self.lineEdit_nome.text().strip()
        email  = self.lineEdit_email.text().strip()
        idade  = self.spinBox_idade.value()         # int
        curso  = self.comboBox_curso.currentText()  # str
        ativo  = self.checkBox_ativo.isChecked()    # bool
        obs    = self.textEdit_obs.toPlainText()    # str multi-linha

        if not nome or not email:
            self.label_status.setText("⚠️ Nome e e-mail são obrigatórios.")
            self.label_status.setStyleSheet("color: #f59e0b;")
            return

        # Inserir linha na QTableWidget
        linha = self.tableWidget_alunos.rowCount()
        self.tableWidget_alunos.insertRow(linha)
        self.tableWidget_alunos.setItem(linha, 0, QTableWidgetItem(nome))
        self.tableWidget_alunos.setItem(linha, 1, QTableWidgetItem(email))
        self.tableWidget_alunos.setItem(linha, 2, QTableWidgetItem(str(idade)))
        self.tableWidget_alunos.setItem(linha, 3, QTableWidgetItem(curso))

        # Atualizar o label de status (objectName: label_status)
        self.label_status.setText(f"✅ {nome} cadastrado com sucesso!")
        self.label_status.setStyleSheet("color: #3fb950;")

        self.limpar_campos()

    def limpar_campos(self):
        """Limpa todos os campos de entrada."""
        self.lineEdit_nome.clear()
        self.lineEdit_email.clear()
        self.spinBox_idade.setValue(14)
        self.comboBox_curso.setCurrentIndex(0)
        self.checkBox_ativo.setChecked(True)
        self.textEdit_obs.clear()
        self.lineEdit_nome.setFocus()

    def validar_formulario(self):
        """Habilita o botão salvar apenas se campos obrigatórios estiverem preenchidos."""
        nome_ok  = bool(self.lineEdit_nome.text().strip())
        email_ok = bool(self.lineEdit_email.text().strip())
        # Habilitar/desabilitar o botão pelo objectName
        self.pushButton_salvar.setEnabled(nome_ok and email_ok)

if __name__ == "__main__":
    app = QApplication(sys.argv)
    janela = CadastroAluno()
    janela.show()
    sys.exit(app.exec())

from PySide6.QtWidgets import QApplication, QDialog, QMainWindow
from PySide6.uic import loadUi

class DialogProduto(QDialog):
    """
    Diálogo criado no Qt Designer e carregado com loadUi.
    Abre em modo modal (bloqueando a janela pai).
    """

    def __init__(self, parent=None, dados_iniciais=None):
        super().__init__(parent)
        loadUi("dialog_produto.ui", self)

        # buttonBox é o objectName padrão do QDialogButtonBox no Designer
        # accepted e rejected são signals do QDialogButtonBox
        self.buttonBox.accepted.connect(self.validar_e_aceitar)
        self.buttonBox.rejected.connect(self.reject)

        # Se estiver editando, pré-preencher os campos
        if dados_iniciais:
            self.lineEdit_produto.setText(dados_iniciais["nome"])
            self.lineEdit_preco.setText(str(dados_iniciais["preco"]))
            self.spinBox_estoque.setValue(dados_iniciais["estoque"])

    def validar_e_aceitar(self):
        """Valida antes de fechar com OK."""
        nome = self.lineEdit_produto.text().strip()
        if not nome:
            # Mostrar erro sem fechar o diálogo
            self.lineEdit_produto.setStyleSheet("border: 1px solid red;")
            self.lineEdit_produto.setPlaceholderText("Campo obrigatório!")
            return  # NÃO chamar accept() — mantém o diálogo aberto
        self.accept()  # fecha com resultado Accepted

    def obter_dados(self):
        """Retorna os dados preenchidos no diálogo."""
        try:
            preco = float(self.lineEdit_preco.text().replace(",", "."))
        except ValueError:
            preco = 0.0
        return {
            "nome":    self.lineEdit_produto.text().strip(),
            "preco":   preco,
            "estoque": self.spinBox_estoque.value()
        }


# ── Janela principal que usa o diálogo ────────────────────────────

class JanelaPrincipal(QMainWindow):
    def __init__(self):
        super().__init__()
        loadUi("janela_principal.ui", self)

        # Conectar botão "Novo Produto" da janela principal
        self.pushButton_novo.clicked.connect(self.abrir_dialogo_novo)
        self.pushButton_editar.clicked.connect(self.abrir_dialogo_editar)

    def abrir_dialogo_novo(self):
        """Abre o diálogo para cadastrar um novo produto."""
        dlg = DialogProduto(parent=self)
        # exec() é MODAL — trava a janela principal até fechar o diálogo
        if dlg.exec() == QDialog.Accepted:
            dados = dlg.obter_dados()
            print(f"Novo produto: {dados}")
            self.statusBar().showMessage(
                f"Produto '{dados['nome']}' cadastrado!", 3000
            )

    def abrir_dialogo_editar(self):
        """Abre o diálogo pre-preenchido para edição."""
        # Dados fictícios — na prática viria do item selecionado na tabela
        dados_existentes = {"nome": "Notebook", "preco": 2500.00, "estoque": 5}

        dlg = DialogProduto(parent=self, dados_iniciais=dados_existentes)
        if dlg.exec() == QDialog.Accepted:
            dados = dlg.obter_dados()
            print(f"Produto editado: {dados}")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    janela = JanelaPrincipal()
    janela.show()
    sys.exit(app.exec())

4. Crie um Dialog separado para cada janela modal complexa

python sistema_notas.py 
"""
Sistema de Notas — Projeto Integrador PySide6
Funcionalidades:
  - Cadastrar alunos com nome e três notas
  - Calcular média automaticamente
  - Listar todos na tabela
  - Editar e excluir registros
  - Salvar/carregar em JSON
"""

import sys
import json
import os
from PySide6.QtWidgets import (
    QApplication, QMainWindow, QWidget,
    QVBoxLayout, QHBoxLayout,
    QLabel, QPushButton, QTableWidget, QTableWidgetItem,
    QDialog, QDialogButtonBox, QFormLayout,
    QLineEdit, QDoubleSpinBox, QMessageBox
)
from PySide6.QtCore import Qt
from PySide6.QtGui import QColor, QFont

ARQUIVO_JSON = "notas.json"

# ══════════════════════════════════════════════════════════════
# Diálogo para adicionar / editar aluno
# ══════════════════════════════════════════════════════════════

class DialogAluno(QDialog):
    def __init__(self, parent=None, dados=None):
        super().__init__(parent)
        self.setWindowTitle("Novo Aluno" if dados is None else "Editar Aluno")
        self.resize(300, 220)

        layout = QVBoxLayout(self)
        form = QFormLayout()

        self.campo_nome = QLineEdit()

        # Três QDoubleSpinBox para as notas
        self.nota1 = QDoubleSpinBox(); self.nota1.setRange(0, 10); self.nota1.setDecimals(1)
        self.nota2 = QDoubleSpinBox(); self.nota2.setRange(0, 10); self.nota2.setDecimals(1)
        self.nota3 = QDoubleSpinBox(); self.nota3.setRange(0, 10); self.nota3.setDecimals(1)

        form.addRow("Nome:",   self.campo_nome)
        form.addRow("Nota 1:", self.nota1)
        form.addRow("Nota 2:", self.nota2)
        form.addRow("Nota 3:", self.nota3)
        layout.addLayout(form)

        botoes = QDialogButtonBox(QDialogButtonBox.Ok | QDialogButtonBox.Cancel)
        botoes.accepted.connect(self._validar)
        botoes.rejected.connect(self.reject)
        layout.addWidget(botoes)

        # Pré-preencher se estiver editando
        if dados:
            self.campo_nome.setText(dados["nome"])
            self.nota1.setValue(dados["nota1"])
            self.nota2.setValue(dados["nota2"])
            self.nota3.setValue(dados["nota3"])

    def _validar(self):
        if not self.campo_nome.text().strip():
            QMessageBox.warning(self, "Atenção", "O nome do aluno é obrigatório.")
            return
        self.accept()

    def obter_dados(self):
        n1 = self.nota1.value()
        n2 = self.nota2.value()
        n3 = self.nota3.value()
        return {
            "nome":  self.campo_nome.text().strip(),
            "nota1": n1,
            "nota2": n2,
            "nota3": n3,
            "media": round((n1 + n2 + n3) / 3, 2)
        }

class SistemaNotas(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("📚 Sistema de Notas — SENAI")
        self.resize(720, 520)
        self.centralizar()

        self.alunos = []  # lista de dicts em memória

        self._criar_ui()
        self._carregar_json()

    def centralizar(self):
        tela = self.screen().availableGeometry()
        self.move(
            (tela.width()  - self.width())  // 2,
            (tela.height() - self.height()) // 2
        )

    def _criar_ui(self):
        central = QWidget()
        self.setCentralWidget(central)
        root = QVBoxLayout(central)
        root.setContentsMargins(16, 16, 16, 16)
        root.setSpacing(12)

        # Título
        titulo = QLabel("Sistema de Controle de Notas")
        titulo.setFont(QFont("Segoe UI", 16, QFont.Bold))
        titulo.setAlignment(Qt.AlignCenter)
        root.addWidget(titulo)

        # Barra de botões
        barra = QHBoxLayout()
        self.btn_novo    = QPushButton("✚ Novo Aluno")
        self.btn_editar  = QPushButton("✏️  Editar")
        self.btn_excluir = QPushButton("🗑  Excluir")
        self.btn_salvar  = QPushButton("💾 Salvar JSON")

        self.btn_editar.setEnabled(False)
        self.btn_excluir.setEnabled(False)

        for b in [self.btn_novo, self.btn_editar, self.btn_excluir, self.btn_salvar]:
            barra.addWidget(b)
        root.addLayout(barra)

        # Tabela
        self.tabela = QTableWidget()
        self.tabela.setColumnCount(6)
        self.tabela.setHorizontalHeaderLabels(
            ["Nome", "Nota 1", "Nota 2", "Nota 3", "Média", "Situação"]
        )
        self.tabela.setSelectionBehavior(QTableWidget.SelectRows)
        self.tabela.setEditTriggers(QTableWidget.NoEditTriggers)
        self.tabela.horizontalHeader().setStretchLastSection(True)
        root.addWidget(self.tabela)

        # Rodapé
        self.label_rodape = QLabel("Nenhum aluno cadastrado.")
        self.label_rodape.setAlignment(Qt.AlignCenter)
        root.addWidget(self.label_rodape)

        # ── Signals ──────────────────────────────────────
        self.btn_novo.clicked.connect(self.novo_aluno)
        self.btn_editar.clicked.connect(self.editar_aluno)
        self.btn_excluir.clicked.connect(self.excluir_aluno)
        self.btn_salvar.clicked.connect(self.salvar_json)
        self.tabela.itemSelectionChanged.connect(self._atualizar_botoes)

    # ── Operações CRUD ────────────────────────────────────

    def novo_aluno(self):
        dlg = DialogAluno(self)
        if dlg.exec() == QDialog.Accepted:
            self.alunos.append(dlg.obter_dados())
            self._atualizar_tabela()

    def editar_aluno(self):
        linha = self.tabela.currentRow()
        if linha < 0: return
        dlg = DialogAluno(self, dados=self.alunos[linha])
        if dlg.exec() == QDialog.Accepted:
            self.alunos[linha] = dlg.obter_dados()
            self._atualizar_tabela()

    def excluir_aluno(self):
        linha = self.tabela.currentRow()
        if linha < 0: return
        nome = self.alunos[linha]["nome"]
        resp = QMessageBox.question(
            self, "Confirmar",
            f"Excluir {nome}?",
            QMessageBox.Yes | QMessageBox.No
        )
        if resp == QMessageBox.Yes:
            self.alunos.pop(linha)
            self._atualizar_tabela()

    # ── Tabela ────────────────────────────────────────────

    def _atualizar_tabela(self):
        self.tabela.setRowCount(0)
        for aluno in self.alunos:
            r = self.tabela.rowCount()
            self.tabela.insertRow(r)
            media = aluno["media"]
            situacao = "Aprovado ✅" if media >= 6 else "Reprovado ❌"
            cor_bg = QColor("#1a3a2a") if media >= 6 else QColor("#3a1a1a")

            valores = [
                aluno["nome"],
                f'{aluno["nota1"]:.1f}',
                f'{aluno["nota2"]:.1f}',
                f'{aluno["nota3"]:.1f}',
                f'{media:.2f}',
                situacao
            ]
            for c, v in enumerate(valores):
                item = QTableWidgetItem(v)
                item.setTextAlignment(Qt.AlignCenter if c > 0 else Qt.AlignLeft | Qt.AlignVCenter)
                item.setBackground(cor_bg)
                self.tabela.setItem(r, c, item)

        total = len(self.alunos)
        aprovados = sum(1 for a in self.alunos if a["media"] >= 6)
        self.label_rodape.setText(
            f"Total: {total} aluno(s) | Aprovados: {aprovados} | Reprovados: {total - aprovados}"
        )

    def _atualizar_botoes(self):
        selecionado = self.tabela.currentRow() >= 0
        self.btn_editar.setEnabled(selecionado)
        self.btn_excluir.setEnabled(selecionado)

    # ── Persistência JSON ─────────────────────────────────

    def salvar_json(self):
        with open(ARQUIVO_JSON, "w", encoding="utf-8") as f:
            json.dump(self.alunos, f, ensure_ascii=False, indent=2)
        self.statusBar().showMessage(f"Salvo em {ARQUIVO_JSON}", 3000)

    def _carregar_json(self):
        if os.path.exists(ARQUIVO_JSON):
            with open(ARQUIVO_JSON, "r", encoding="utf-8") as f:
                self.alunos = json.load(f)
            self._atualizar_tabela()


if __name__ == "__main__":
    app = QApplication(sys.argv)
    janela = SistemaNotas()
    janela.show()
    sys.exit(app.exec())
